function(hybridrt_configure_target target)
    add_library(hrt::${target} ALIAS ${target})
    target_include_directories(
        ${target}
        PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
               $<BUILD_INTERFACE:${PROJECT_BINARY_DIR}/include>
        PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(
        ${target}
        PROPERTIES POSITION_INDEPENDENT_CODE ON
                   CXX_VISIBILITY_PRESET hidden
                   VISIBILITY_INLINES_HIDDEN ON)
    hybridrt_apply_warnings(${target})
endfunction()

if(APPLE)
    enable_language(OBJCXX)
    target_sources(hybridrt PRIVATE preprocess/metal_image_preprocessor.cpp)
    set_source_files_properties(preprocess/metal_image_preprocessor.cpp PROPERTIES
        LANGUAGE OBJCXX COMPILE_OPTIONS "-fobjc-arc")
    # The Metal kernel is compiled at run time from its source, embedded here; editing it reconfigures.
    set(_metal_kernel ${CMAKE_CURRENT_SOURCE_DIR}/backend/metal/kernels/image_kernel.metal)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_metal_kernel})
    file(READ ${_metal_kernel} HRT_METAL_IMAGE_KERNEL)
    file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/metal_image_kernel.cpp CONTENT [[
// Generated by source/CMakeLists.txt from source/backend/metal/kernels/image_kernel.metal; do not edit.
#include "include/backend/metal/kernels/image_kernel.hpp"

#if defined(__clang__)
#pragma clang diagnostic ignored "-Woverlength-strings"
#endif

namespace hrt::backend {

auto metal_image_kernel() -> const char* {
  return R"metal(@HRT_METAL_IMAGE_KERNEL@)metal";
}

} // namespace hrt::backend
]] @ONLY)
    target_sources(hybridrt PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/metal_image_kernel.cpp)
    set_target_properties(hybridrt PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)

    target_compile_definitions(hybridrt PRIVATE HRT_USE_METAL)
